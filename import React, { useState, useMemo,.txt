  const extractSkillsFromResume = (text) => {

      // Escape special regex characters like '.' and '+'
      const escapedSkill = skill.replace(/\./g, '\\.').replace(/\+/g, '\\+').replace('c++', 'c\\+\\+'); // Fix for c++

        // Find the "canonical" version of the skill
        const canonicalSkill = MOCK_JOB_OPENINGS

        const finalSkill = canonicalSkill === skill

        foundSkills.add(finalSkill);

   * Simulates the AI recommendation engine.
   */
  const runRecommendationEngine = () => {
    setIsLoading(true);

    // 1. Consolidate all student skills
    const resumeSkills = extractSkillsFromResume(studentProfile.resumeText);
    const allStudentSkills = [
      ...new Set(

      // 3. Compare skills and generate matches

        const required = job.requiredSkills.map(s => s.toLowerCase());

          : 0;

        return {
          ...job,
          matchingSkills: job.requiredSkills.filter(skill => allStudentSkills.includes(skill.toLowerCase())),

          matchScore: Math.round(matchScore),

        const companyTypePass = (() => {
          if (selectedCompanyTypes.length === 0) return true; // If none selected, pass all
          return selectedCompanyTypes.includes(match.companyType);
        })();

        return locationPass && companyTypePass; // Must pass both filters
      });

      // 5. Generate Skill Upgrade Suggestions

      // Re-calculate gaps based ONLY on filtered jobs
      const filteredGaps = new Map();
      filteredMatches.forEach(job => {
        job.skillGaps.forEach(gap => {
          filteredGaps.set(gap, (filteredGaps.get(gap) || 0) + 1);
        });
      });

      const priorityGaps = Array.from(filteredGaps.entries())
        .sort((a, b) => b[1] - a[1]) // Sort by frequency
        .slice(0, 3) // Get top 3
        .map(item => item[0]); // Get skill name

      const emergingSuggestions = new Set();
      allStudentSkills.forEach(skill => {
        const suggestion = EMERGING_SKILL_MAP[skill];
        // THE FIX: Only add if it exists AND user doesn't already have it
        if (suggestion && !allStudentSkills.includes(suggestion.toLowerCase())) {
          emergingSuggestions.add(suggestion);
        }
      });

      // 6. Sort recommendations by match score
      filteredMatches.sort((a, b) => b.matchScore - a.matchScore);

      setRecommendations(filteredMatches);
      setUpgradeSuggestions({
        priority: priorityGaps,
        emerging: Array.from(emergingSuggestions)
      });
      setIsLoading(false);
      setActiveTab('recommendations');
    }, 1500); // 1.5 second delay
  };

  // --- Gemini API Call Functions ---

  /**
   * Generic function to call the Gemini API with exponential backoff.
   */
  const callGeminiApi = async (systemPrompt, userQuery, useGrounding = false, maxRetries = 3) => {
    const apiKey = ""; // API key will be injected by the environment
    const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;

    const payload = {
      contents: [{ parts: [{ text: userQuery }] }],
      systemInstruction: {
        parts: [{ text: systemPrompt }]
      },
    };

    if (useGrounding) {
      payload.tools = [{ "google_search": {} }];
    }

    let delay = 1000; // Start with 1 second
    for (let i = 0; i <= maxRetries; i++) {
      try {
        const response = await fetch(apiUrl, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });

        if (!response.ok) {
          if (response.status === 429 || response.status >= 500) {
            // Throttling or server error, wait and retry
            if (i === maxRetries) {
              return { error: `API call failed after ${maxRetries} retries with status: ${response.status}` };
            }
            await new Promise(res => setTimeout(res, delay));
            delay *= 2; // Exponential backoff
            continue; // Retry
          }
          return { error: `API call failed with status: ${response.status}` };
        }

        const result = await response.json();
        const candidate = result.candidates?.[0];

        if (candidate && candidate.content?.parts?.[0]?.text) {
          const text = candidate.content.parts[0].text;
          let sources = [];
          const groundingMetadata = candidate.groundingMetadata;
          if (groundingMetadata && groundingMetadata.groundingAttributions) {
            sources = groundingMetadata.groundingAttributions
              .map(attribution => ({
                uri: attribution.web?.uri,
                title: attribution.web?.title,
              }))
              .filter(source => source.uri && source.title);
          }
          return { text, sources };
        } else {
          return { error: "API response was not as expected." };
        }
      } catch (error) {
        if (i === maxRetries) {
          return { error: `API call failed: ${error.message}` };
        }
        await new Promise(res => setTimeout(res, delay));
        delay *= 2;
      }
    }
    return { error: "An unknown error occurred during the API call." };
  };

  // Helper: Base64 to ArrayBuffer
  const base64ToArrayBuffer = (base64) => {
    const binaryString = window.atob(base64);
    const len = binaryString.length;
    const bytes = new Uint8Array(len);
    for (let i = 0; i < len; i++) {
      bytes[i] = binaryString.charCodeAt(i);
    }
    return bytes.buffer;
  };

  // Helper: PCM to WAV
  const pcmToWav = (pcmData, sampleRate) => {
    const numChannels = 1;
    const bitsPerSample = 16;
    const bytesPerSample = bitsPerSample / 8;
    const blockAlign = numChannels * bytesPerSample;
    const byteRate = sampleRate * blockAlign;
    const dataSize = pcmData.length * bytesPerSample;
    const fileSize = 44 + dataSize;

    const buffer = new ArrayBuffer(fileSize);
    const view = new DataView(buffer);

    // RIFF header
    view.setUint32(0, 0x52494646, false); // "RIFF"
    view.setUint32(4, fileSize - 8, true); // File size - 8
    view.setUint32(8, 0x57415645, false); // "WAVE"

    // fmt chunk
    view.setUint32(12, 0x666d7420, false); // "fmt "
    view.setUint32(16, 16, true); // Subchunk1 size (16 for PCM)
    view.setUint16(20, 1, true); // Audio format (1 for PCM)
    view.setUint16(22, numChannels, true); // Num channels
    view.setUint32(24, sampleRate, true); // Sample rate
    view.setUint32(28, byteRate, true); // Byte rate
    view.setUint16(32, blockAlign, true); // Block align
    view.setUint16(34, bitsPerSample, true); // Bits per sample

    // data chunk
    view.setUint32(36, 0x64617461, false); // "data"
    view.setUint32(40, dataSize, true); // Subchunk2 size

    // Write PCM data
    let offset = 44;
    for (let i = 0; i < pcmData.length; i++, offset += 2) {
      view.setInt16(offset, pcmData[i], true);
    }

    return new Blob([buffer], { type: 'audio/wav' });
  };

  // Main TTS API call function
  const callTtsApi = async (text) => {
    const apiKey = ""; // API key will be injected by the environment
    const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-tts:generateContent?key=${apiKey}`;

    const payload = {
      contents: [{
        parts: [{ text: `Say this in a helpful, professional tone: ${text}` }]
      }],
      generationConfig: {
        responseModalities: ["AUDIO"],
        speechConfig: {
          voiceConfig: {
            prebuiltVoiceConfig: { voiceName: "Kore" } // A professional voice
          }
        }
      },
      model: "gemini-2.5-flash-preview-tts"
    };

    try {
      const response = await fetch(apiUrl, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });
      if (!response.ok) throw new Error(`TTS API failed with status: ${response.status}`);
      const result = await response.json();
      const part = result?.candidates?.[0]?.content?.parts?.[0];
      const audioData = part?.inlineData?.data;
      const mimeType = part?.inlineData?.mimeType;

      if (audioData && mimeType && mimeType.startsWith("audio/L16")) {
        const sampleRateMatch = mimeType.match(/rate=(\d+)/);
        const sampleRate = sampleRateMatch ? parseInt(sampleRateMatch[1], 10) : 24000;
        const pcmData = base64ToArrayBuffer(audioData);
        const pcm16 = new Int16Array(pcmData);
        const wavBlob = pcmToWav(pcm16, sampleRate);
        return { wavBlob };
      } else {
        throw new Error("Invalid TTS response format.");
      }
    } catch (error) {
      console.error("Error generating TTS:", error);
      return { error: error.message };
    }
  };

          <svg className="animate-spin h-12 w-12 text-indigo-600" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24">
            <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4"></circle>
            <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path>
          </svg>
          <h2 className="mt-4 text-xl font-semibold text-gray-700">Running AI Analysis...</h2>
          <p className="text-gray-500">Finding the best matches for your profile.</p>
        </div>
      );
    }

    if (recommendations.length === 0) {
      return (
        <div className="text-center p-10 bg-white rounded-lg shadow-lg">
          <h2 className="text-xl font-semibold text-gray-700">No Recommendations Yet</h2>
          <p className="text-gray-500 mt-2">Go to the 'Profile' tab, fill in your details, and click 'Analyze Profile & Find Matches' to get started.</p>
        </div>
      );
    }
    return (
      <div className="space-y-4">
        <h2 className="text-2xl font-semibold text-gray-800 mb-4">Your Top Recommendations</h2>
        {/* Render Skill Upgrade Suggestions first */}
        {renderSkillUpgradeView()}
        
        {recommendations.map(job => (
          <div key={job.id} className="bg-white p-6 rounded-lg shadow-lg">
            <div className="flex justify-between items-start">
              <div>
                <h3 className="text-lg font-semibold text-indigo-600">{job.title}</h3>
                <p className="text-md font-medium text-gray-700">{job.company} - <span className="text-gray-500 font-normal">{job.location.city}, {job.location.country}</span></p>
              </div>
              <div className="text-right flex-shrink-0 ml-4">
                <span className="font-bold text-xl text-green-600">{job.matchScore}% Match</span>
                <span className="block px-3 py-1 rounded-full text-sm font-medium bg-gray-100 text-gray-800 mt-1">
                  {job.companyType}
                </span>
              </div>
            </div>
            <p className="mt-2 text-sm text-gray-600">{job.description}</p>
            
            <div className="mt-4">
              <h4 className="text-sm font-semibold text-gray-700">Skill Analysis</h4>
              <div className="flex flex-wrap mt-2">
                {job.matchingSkills.map(skill => (
                  <SkillTag key={skill} skill={skill} type="match" />
                ))}
                {job.skillGaps.map(skill => (
                  <SkillTag key={skill} skill={skill} type="gap" />
                ))}
              </div>
            </div>

            <div className="mt-4 flex flex-wrap gap-2">
              <button
                onClick={() => handleVibeCheck(job.company)}
                className="px-3 py-1 rounded-full text-xs font-medium bg-indigo-100 text-indigo-700 hover:bg-indigo-200 shadow-sm"
              >
                Vibe Check ✨
              </button>
              <button
                onClick={() => handleInterviewPrep(job)}
                className="px-3 py-1 rounded-full text-xs font-medium bg-indigo-100 text-indigo-700 hover:bg-indigo-200 shadow-sm"
              >
                Prep for Interview ✨
              </button>
              <button
                onClick={() => handleDraftPitch(job)}
                className="px-3 py-1 rounded-full text-xs font-medium bg-indigo-100 text-indigo-700 hover:bg-indigo-200 shadow-sm"
              >
                Draft Pitch ✨
              </button>
              <button
                onClick={() => handleApply(job)}
                className="px-3 py-1 rounded-full text-xs font-medium bg-indigo-600 text-white hover:bg-indigo-700 shadow-sm"
              >
                Apply Now
              </button>
            </div>

          </div>
        ))}
      </div>
    );
  };

  const renderLoginView = () => (
    <div className="min-h-screen bg-gray-100 flex items-center justify-center">
      <div className="max-w-md w-full bg-white p-8 rounded-lg shadow-xl">
        <div className="text-center">
          <svg className="h-12 w-12 text-indigo-600 mx-auto" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24" stroke="currentColor">
            <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M21 13.255A23.931 23.931 0 0112 15c-3.183 0-6.22-.62-9-1.745M16 6V4a2 2 0 00-2-2h-4a2 2 0 00-2 2v2m4 6h.01M5 20h14a2 2 0 002-2V8a2 2 0 00-2-2H5a2 2 0 00-2 2v10a2 2 0 002 2z" />
          </svg>
          <h2 className="mt-4 text-center text-3xl font-extrabold text-gray-900">
            Welcome to TalentNexus
          </h2>
          <p className="mt-2 text-center text-sm text-gray-600">
            Your AI-powered internship finder
          </p>
        </div>
        <form className="mt-8 space-y-6" onSubmit={handleLogin}>
          {/* This is a mock login. In a real app, you'd have inputs. */}
          <div className="rounded-md shadow-sm -space-y-px">
            <div>
              <label htmlFor="email-address" className="sr-only">Email address</label>
              <input id="email-address" name="email" type="email" autoComplete="email" required
                className="appearance-none rounded-none relative block w-full px-3 py-2 border border-gray-300 placeholder-gray-500 text-gray-900 rounded-t-md focus:outline-none focus:ring-indigo-500 focus:border-indigo-500 focus:z-10 sm:text-sm"
                placeholder="Email address (e.g., student@email.com)"
                defaultValue="student@email.com"
              />
            </div>
            <div>
              <label htmlFor="password" className="sr-only">Password</label>
              <input id="password" name="password" type="password" autoComplete="current-password" required
                className="appearance-none rounded-none relative block w-full px-3 py-2 border border-gray-300 placeholder-gray-500 text-gray-900 rounded-b-md focus:outline-none focus:ring-indigo-500 focus:border-indigo-500 focus:z-10 sm:text-sm"
                placeholder="Password"
                defaultValue="password"
              />
            </div>
          </div>
          <button type="submit"
            className="group relative w-full flex justify-center py-2 px-4 border border-transparent text-sm font-medium rounded-md text-white bg-indigo-600 hover:bg-indigo-700 focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-indigo-500"
          >
            Sign in
          </button>
        </form>
      </div>
    </div>
  );

  // --- Main App Return ---
  if (!isLoggedIn) {
    return renderLoginView();
  }

  return (
    <>
      {/* Notification */}
      {renderNotification()}

      {/* Main App UI */}
      <div className="min-h-screen bg-gray-100">
        {renderHeader()}
        <main>
          <div className="max-w-7xl mx-auto py-6 sm:px-6 lg:px-8">
            {activeTab === 'profile' && renderProfileView()}
            {activeTab === 'jobs' && renderJobsView()}
            {activeTab === 'recommendations' && renderRecommendationsView()}
            {activeTab === 'inbox' && renderInboxView()}
          </div>
        </main>
      </div>

      {/* Modal */}
      {modalContent.title && (
        <Modal
          title={modalContent.title}
          content={modalContent.content}
          isLoading={isApiLoading}
          onClose={() => {
            setModalContent({ title: '', content: '' });
            if (audioSource) audioSource.stop(); // Stop audio on close
            setAudioBuffer(null);
            setIsAudioPlaying(false);
          }}
          onAction={modalContent.onAction}
          actionLabel={modalContent.actionLabel}
          onTtsAction={handlePlayTts}
          showAudioPlayer={modalContent.showAudioPlayer}
          isAudioPlaying={isAudioPlaying}
        />
      )}

      {/* Apply Modal */}
      {renderApplyModal()}

      {/* Upgrade Modal */}
      {renderUpgradeModal()}

      {/* Audio Player Element (hidden) */}
      <audio 
        ref={el => { if (el && !audioContext) setAudioContext(new (window.AudioContext || window.webkitAudioContext)()) }} 
      />
    </>
  );
}
